  <script>
    // Handle Affordability Calculation
    document.getElementById('affordabilityForm').addEventListener('submit', async (e) => {
      e.preventDefault();
      const resBox = document.getElementById('affordabilityResult');
      
      const payload = {
        netMonthlyPay: document.getElementById('netPay').value,
        existingDeductions: document.getElementById('existingDeductions').value,
        proposedInstallment: document.getElementById('proposedInstallment').value
      };

      try {
        const res = await fetch('/api/v1/loans/check-affordability', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
        const data = await res.json();
        
        if (data.success) {
          const aff = data.affordability;
          const statusBadge = aff.isAffordable 
            ? '<span class="badge badge-pass p-2 fs-6">APPROVED (Within 40% Cap)</span>'
            : '<span class="badge badge-fail p-2 fs-6">REJECTED (Exceeds 40% Cap)</span>';

          resBox.className = 'mt-4 alert ' + (aff.isAffordable ? 'alert-success' : 'alert-danger');
          resBox.innerHTML = `
            <h5>Status: ${statusBadge}</h5>
            <ul class="mb-0 mt-2">
              <li><strong>Max Allowable Deductions (40%):</strong> ZMW ${aff.maxAllowableDeductions}</li>
              <li><strong>Total Deductions After Loan:</strong> ZMW ${aff.totalFutureDeductions}</li>
              <li><strong>Remaining Borrowing Capacity:</strong> ZMW ${aff.availableCap}</li>
            </ul>
          `;
        }
      } catch (err) {
        alert('Error testing affordability.');
      }
    });

    // Handle Schedule Calculation
    document.getElementById('scheduleForm').addEventListener('submit', async (e) => {
      e.preventDefault();
      const resBox = document.getElementById('scheduleResult');

      const payload = {
        principal: document.getElementById('loanPrincipal').value,
        interestRatePerMonth: document.getElementById('interestRate').value,
        tenureMonths: document.getElementById('tenureMonths').value
      };

      try {
        const res = await fetch('/api/v1/loans/calculate', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
        const data = await res.json();

        if (data.success) {
          const calc = data.loanCalculation;
          let tableHtml = `
            <div class="alert alert-info">
              <div><strong>Total Interest:</strong> ZMW ${calc.totalInterest}</div>
              <div><strong>Total Repayment:</strong> ZMW ${calc.totalRepayment}</div>
              <div><strong>Monthly Installment:</strong> ZMW ${calc.monthlyInstallment}</div>
            </div>
            <table class="table table-sm table-striped">
              <thead>
                <tr>
                  <th>#</th>
                  <th>Due Date</th>
                  <th>Principal</th>
                  <th>Interest</th>
                  <th>Total</th>
                </tr>
              </thead>
              <tbody>
          `;

          calc.schedule.forEach(item => {
            tableHtml += `
              <tr>
                <td>${item.installmentNumber}</td>
                <td>${item.dueDate}</td>
                <td>${item.principalDue}</td>
                <td>${item.interestDue}</td>
                <td>${item.totalInstallment}</td>
              </tr>
            `;
          });

          tableHtml += `</tbody></table>`;
          resBox.className = 'mt-4 d-block';
          resBox.innerHTML = tableHtml;
        }
      } catch (err) {
        alert('Error generating schedule.');
      }
    });
  </script>
